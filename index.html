const SHEET_NAME = 'Data'; const HEADERS = ['ClientName','DisplayName','Block','Week','Squat','ChinReps','ChinLoad','Abs','HipThrust','Row','OHP','Bench','LastUpdated']; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.appendRow(HEADERS); } return sheet; } /** * ONE-TIME SETUP STEP — run this manually, once, from the Apps Script editor. * Adds the Bench column before LastUpdated. Safe to run more than once — * checks first and does nothing if already migrated. * * How to run it: paste this whole file in, save (Cmd+S), then use the * runMigrationNow() helper at the very bottom the same way as last time. */ function migrateAddBenchColumn_() { const sheet = getSheet_(); const lastCol = sheet.getLastColumn(); const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0]; if (headers.indexOf('Bench') !== -1) { Logger.log('Already migrated — nothing to do.'); return; } sheet.insertColumnBefore(lastCol); // inserts before LastUpdated, shifting it right sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]); Logger.log('Bench column added.'); } function normalizeName_(name) { return String(name || '').trim().replace(/\s+/g, ' ').toLowerCase(); } function doGet(e) { const sheet = getSheet_(); const action = e.parameter.action; const vals = sheet.getDataRange().getValues(); if (action === 'list') { const names = []; for (let i = 1; i < vals.length; i++) { const n = vals[i][1] || vals[i][0]; if (n && names.indexOf(n) === -1) names.push(n); } return jsonOut_({ names: names }); } if (action === 'get') { const name = normalizeName_(e.parameter.name); const data = blankData_(); for (let i = 1; i < vals.length; i++) { const row = vals[i]; if (normalizeName_(row[0]) !== name) continue; const blockNum = row[2]; const weekNum = row[3]; const bIdx = blockNum - 1; const wIdx = weekNum - ((bIdx) * 5) - 1; if (bIdx < 0 || bIdx > 4 || wIdx < 0 || wIdx > 4) continue; data.blocks[bIdx].weeks[wIdx] = { squat: row[4] || '', chin: row[5] || '', hipthrust: row[8] || '', row: row[9] || '', ohp: row[10] || '', bench: row[11] || '' }; if (row[6]) data.blocks[bIdx].chinLoad = row[6]; } return jsonOut_({ data: data }); } return jsonOut_({ error: 'unknown action' }); } function doPost(e) { const lock = LockService.getScriptLock(); const gotLock = lock.tryLock(30000); if (!gotLock) { return jsonOut_({ ok: false, error: 'Server busy, try again.' }); } try { const sheet = getSheet_(); let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonOut_({ ok: false, error: 'Malformed request body.' }); } const rawName = body.name; const displayName = body.displayName || body.name; const data = body.data; if (!rawName || !data || !Array.isArray(data.blocks)) { return jsonOut_({ ok: false, error: 'missing name or data' }); } const name = normalizeName_(rawName); const vals = sheet.getDataRange().getValues(); const rowIndex = {}; for (let i = 1; i < vals.length; i++) { const key = normalizeName_(vals[i][0]) + '|' + vals[i][2] + '|' + vals[i][3]; rowIndex[key] = i + 1; } const now = new Date(); data.blocks.forEach((block) => { block.weeks.forEach((wk, wIdx) => { const weekNum = ((block.block - 1) * 5) + wIdx + 1; const key = name + '|' + block.block + '|' + weekNum; const existingRowNum = rowIndex[key]; // Preserve any old Abs value already sitting in the sheet — the // app no longer edits this column, but we don't wipe it either. const existingAbs = existingRowNum ? (vals[existingRowNum - 1][7] || '') : ''; const rowValues = [ name, displayName, block.block, weekNum, wk.squat || '', wk.chin || '', block.chinLoad || '', existingAbs, wk.hipthrust || '', wk.row || '', wk.ohp || '', wk.bench || '', now ]; if (existingRowNum) { sheet.getRange(existingRowNum, 1, 1, rowValues.length).setValues([rowValues]); } else { sheet.appendRow(rowValues); rowIndex[key] = sheet.getLastRow(); } }); }); return jsonOut_({ ok: true }); } finally { lock.releaseLock(); } } function blankData_() { return { blocks: [1,2,3,4,5].map(b => ({ block: b, chinLoad: "", weeks: [1,2,3,4,5].map(() => ({squat:"",chin:"",hipthrust:"",row:"",ohp:"",bench:""})) })) }; } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function runMigrationNow() { migrateAddBenchColumn_(); }
